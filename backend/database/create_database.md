CREATE DATABASE organizer_de_tareas;\n\nUSE organizer_de_tareas;\n\nCREATE TABLE tasks (\n  id INT AUTO_INCREMENT,\n  title VARCHAR(255),\n  description TEXT,\n  completed BOOLEAN DEFAULT FALSE,\n  PRIMARY KEY (id)\n);\n\nCREATE TABLE users (\n  id INT AUTO_INCREMENT,\n  username VARCHAR(255),\n  email VARCHAR(255),\n  password VARCHAR(255),\n  PRIMARY KEY (id)\n);\n\nCREATE TABLE projects (\n  id INT AUTO_INCREMENT,\n  title VARCHAR(255),\n  description TEXT,\n  PRIMARY KEY (id)\n);\n
